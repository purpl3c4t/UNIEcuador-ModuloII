### Fundamento Técnico y Objetivo

El objetivo es comprobar cómo el malware implementa salvaguardas condicionales (*Execution Guardrails*, técnica MITRE T1627) mediante sondeo de la CPU (`CPUID`), validación de hardware y latencias `RDTSC` para detectar entornos de sandboxing. Se compara la efectividad de los detectores en User-Mode frente a la **Introspección de Máquinas Virtuales (VMI)** transparente asistida por hardware (como DRAKVUF con EPT).

### Script 
 `guardrails_sandbox.py`

```
#!/usr/bin/env python3
"""
Lab 3: guardrails_sandbox.py
Inyección de Execution Guardrails y exportación de telemetría de 120 segundos.
"""
import time
import json
import os
import sys

class ExecutionGuardrails:
    def __init__(self, is_sandbox_env: bool = False, vcpu_count: int = 1):
        self.is_sandbox_env = is_sandbox_env
        self.vcpu_count = vcpu_count

    def check_hypervisor_cpuid(self) -&gt; bool:
        if self.is_sandbox_env:
            print("  [!] CPUID check: Detectada firma de hypervisor 'VMwareVMware'.")
            return True
        return False

    def check_hardware_resources(self) -&gt; bool:
        if self.vcpu_count &lt;= 2:
            print(f"  [!] Resource check: Entorno de análisis con {self.vcpu_count} vCPUs.")
            return True
        return False

    def timing_evasion_rdtsc(self, simulated_hook_latency: bool = False) -&gt; bool:
        start = time.perf_counter()
        if simulated_hook_latency:
            time.sleep(0.01) # Sobrecarga de API Hooking tradicional
        elapsed = time.perf_counter() - start
        if elapsed &gt; 0.005:
            print(f"  [!] RDTSC check: Latencia anómala por inspección ({elapsed*1000:.2f} ms).")
            return True
        return False

def simulate_detonation(guardrails: ExecutionGuardrails, use_hypervisor_vmi: bool = False):
    telemetry_events = []
    is_vm = guardrails.check_hypervisor_cpuid()
    is_low_res = guardrails.check_hardware_resources()
    timing_anomaly = guardrails.timing_evasion_rdtsc(simulated_hook_latency=(not use_hypervisor_vmi))

    if is_vm or is_low_res or timing_anomaly:
        print("  [X] Execution Guardrails activados -&gt; Desviando a exit(0)...")
        telemetry_events.append({"timestamp": 1, "api": "ExitProcess", "category": "termination", "suspicious": False})
        return telemetry_events, False

    print("  [+] Guardrails superados -&gt; Ejecutando carga útil en memoria...")
    actions = [
        (5, "NtAllocateVirtualMemory", "memory", True),
        (12, "NtWriteVirtualMemory", "memory", True),
        (25, "NtProtectVirtualMemory", "memory", True),
        (45, "RegQueryValueExW", "persistence", False),
        (70, "NtCreateThreadEx", "injection", True),
        (105, "NtDelayExecution", "anti-analysis", False),
        (120, "NtTerminateProcess", "process", False)
    ]
    for t_sec, api, cat, is_susp in actions:
        telemetry_events.append({"timestamp": t_sec, "api": api, "category": cat, "suspicious": is_susp})
    return telemetry_events, True

if __name__ == "__main__":
    print("=== Laboratorio 3: Simulación de Guardrails y Evasión Dinámica ===")
    
    print("\n--- Caso A: Sandbox Tradicional con Hooking (2 vCPUs) ---")
    g_sand = ExecutionGuardrails(is_sandbox_env=True, vcpu_count=2)
    ev_a, exec_a = simulate_detonation(g_sand, use_hypervisor_vmi=False)
    print(f"Resultado Caso A: {'Payload Ejecutado' if exec_a else 'Evasión Exitosa (Payload Oculto)'}")

    print("\n--- Caso B: Sandbox con Hypervisor VMI Transparente (8 vCPUs) ---")
    g_target = ExecutionGuardrails(is_sandbox_env=False, vcpu_count=8)
    ev_b, exec_b = simulate_detonation(g_target, use_hypervisor_vmi=True)
    print(f"Resultado Caso B: {'Payload Ejecutado' if exec_b else 'Evasión Exitosa'}")

    # Exportación del archivo para el Laboratorio 4
    output_log = "telemetry_120s.json"
    with open(output_log, "w") as f:
        json.dump(ev_b, f, indent=2)
    print(f"\n[+] Telemetría de 120s exportada: {output_log}")

```
